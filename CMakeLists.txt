cmake_minimum_required(VERSION 3.24)
project(OpenMSTestData VERSION 1.0.0 LANGUAGES NONE)
include(CMakePackageConfigHelpers)
include(cmake/OpenMS4Dependencies.cmake)
# Installed harnesses carry package identity without requiring a Git checkout.
if(NOT OPENMS4_SOURCE_REVISION AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/OpenMSTestDataSource.cmake")
  include("${CMAKE_CURRENT_SOURCE_DIR}/OpenMSTestDataSource.cmake")
endif()
openms4_source_revision(OpenMSTestData_SOURCE_REVISION OpenMSTestData_SOURCE_DIRTY)
set(OpenMSTestData_REGRESSION_SOURCE_DIR share/openms4-test-data/${PROJECT_VERSION})
set(OpenMSTestData_TOPP_DIR ${OpenMSTestData_REGRESSION_SOURCE_DIR}/topp)
configure_package_config_file(cmake/OpenMSTestDataConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/OpenMSTestDataConfig.cmake
  INSTALL_DESTINATION lib/cmake/OpenMSTestData PATH_VARS OpenMSTestData_TOPP_DIR OpenMSTestData_REGRESSION_SOURCE_DIR)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/OpenMSTestDataConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion ARCH_INDEPENDENT)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/OpenMSTestDataSource.cmake"
  "set(OPENMS4_SOURCE_REVISION \"${OpenMSTestData_SOURCE_REVISION}\")\nset(OPENMS4_SOURCE_DIRTY \"${OpenMSTestData_SOURCE_DIRTY}\")\n")
install(DIRECTORY topp/ DESTINATION ${OpenMSTestData_TOPP_DIR})
install(FILES CMakeLists.txt dependencies.lock.json License.txt
  "${CMAKE_CURRENT_BINARY_DIR}/OpenMSTestDataSource.cmake"
  DESTINATION ${OpenMSTestData_REGRESSION_SOURCE_DIR})
install(DIRECTORY cmake/ DESTINATION ${OpenMSTestData_REGRESSION_SOURCE_DIR}/cmake)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/OpenMSTestDataConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/OpenMSTestDataConfigVersion.cmake DESTINATION lib/cmake/OpenMSTestData)
option(OPENMS4_REGRESSION_TESTS "Register original suite against installed tool binaries" OFF)
if(OPENMS4_REGRESSION_TESTS)
  include(CTest)
  openms4_find_core_data(COMPONENTS TestSupport)
  set(OPENMS4_TOOLS_BIN "" CACHE PATH "Directory containing the installed tool binaries")
  set(OPENMS4_TOOL_NAMES "" CACHE STRING "Explicit tool names when installed product manifests are unavailable")
  if(WIN32)
    set(OPENMS4_EXECUTABLE_SUFFIX ".exe")
  else()
    set(OPENMS4_EXECUTABLE_SUFFIX "")
  endif()
  if(NOT EXISTS "${OPENMS4_TOOLS_BIN}/FuzzyDiff${OPENMS4_EXECUTABLE_SUFFIX}")
    message(FATAL_ERROR "Set OPENMS4_TOOLS_BIN to a complete installed suite (FuzzyDiff required)")
  endif()
  set(TOPP_TOOLS ${OPENMS4_TOOL_NAMES})
  if(NOT TOPP_TOOLS)
    file(GLOB _manifests "${OPENMS4_TOOLS_BIN}/../share/openms4/tools/*.tools.tsv")
    foreach(_manifest IN LISTS _manifests)
      file(STRINGS "${_manifest}" _entries REGEX "^[A-Za-z][A-Za-z0-9_]*\t")
      foreach(_entry IN LISTS _entries)
        string(REGEX MATCH "^[A-Za-z][A-Za-z0-9_]*" _tool "${_entry}")
        list(APPEND TOPP_TOOLS "${_tool}")
      endforeach()
    endforeach()
  endif()
  list(REMOVE_DUPLICATES TOPP_TOOLS)
  list(SORT TOPP_TOOLS)
  if(NOT TOPP_TOOLS)
    message(FATAL_ERROR "Install product manifests in share/openms4/tools or set OPENMS4_TOOL_NAMES explicitly")
  endif()
  foreach(_tool IN LISTS TOPP_TOOLS)
    if(NOT _tool MATCHES "^[A-Za-z][A-Za-z0-9_]*$" OR NOT EXISTS "${OPENMS4_TOOLS_BIN}/${_tool}${OPENMS4_EXECUTABLE_SUFFIX}")
      message(FATAL_ERROR "Installed tool ${_tool} is missing from OPENMS4_TOOLS_BIN")
    endif()
  endforeach()
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${OPENMS4_TOOLS_BIN}")
  option(WITH_GUI "Enable desktop numerical tests when desktop products are installed" OFF)
  option(HAS_XSERVER "Enable tests that require a graphical display" OFF)
  set(WITH_WNETALIGN ${OpenMS_WITH_WNETALIGN})
  set(WITH_OPENTIMS ${OpenMS_WITH_OPENTIMS})
  set(WITH_THERMO_RAW ${OpenMS_WITH_THERMO_RAW})
  set(DISABLE_OPENSWATH OFF)
  if(NOT OpenMS_WITH_OPENSWATH)
    set(DISABLE_OPENSWATH ON)
  endif()
  add_subdirectory(topp)
endif()
